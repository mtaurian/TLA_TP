<script>
    const isMultipleOf = (value, of) => {
        return (value % of) === 0;
    }

    const isDivisorOf = (divisor, of) => {
        return (of % divisor) === 0;
    }

    const like = (value, regex) => {
        return regex.test(value);
    }

    const lengthEquals = (str1, str2) => {
        return str1.length === str2.length;
    }

    const longerThan = (str1, value) => {
        return str1.length > value;
    }

    const shorterThan = (str1, value) => {
        return str1.length < value;
    }

    const isEmpty = (str) => {
        return str.length == 0;
    }

    const isBefore = (date1, date2) => {
        return new Date(date1) < new Date(date2);
    }

    const isAfter = (date1, date2) => {
        return new Date(date1) > new Date(date2);
    }

    const isWeekend = (date) => {
        const day = new Date(date).getDay();
        return day === 5 || day === 6;
    }


</script>
