#Title "The Form Title"
#Closure "Thanks for submiting the form"
@FormConfig {
	#SubmitBtnText "Submit"
	#SafeAndSound
	#Theme _TTPD
}
@Step s1 {
	#Title "Step 1: Data"
	#Description "Fill your personal Data"
	@GetAwayCar {
		WHEN howOld isLowerThan 13 GOTO END
		WHEN howOld isGreaterThan 18 OR howOld isEmpty GOTO s3 
	}
	@Section {
		#Title "First Section of s1"
		#Description "First Section Desc"
		@ShowIf isOlderThan18 { howOld isGreaterThan 18 }

		@Question name {
			#Type TEXT
			#Title "Name"
			#PlaceHolder "Enter your name"
			#Default "."
			#Required
			#Help "your name goes here"
			@Glitch {
				@Error {
					#Message "Too long"
					@ShowIf { name isLongerThan 20 }
				}
				@Error {
					#Message "Too short"
					@ShowIf { name isShorterThan 1 OR name isEmpty }
				}
			}
		}

		@Question howOld {
			#Type NUMERIC
			#Title "How old are you?"
			#Required		
		}

		@Question favoriteBeer {
			#Type SELECT TEXT
			#Title "What's your favorite beer?"
			#Options ["Brahma", "Patagonia", "Heineken" @ShowIf { howOld isGreaterThan 21 }]
			@ShowIf(isOlderThan18)
		}		
	}
	@Section {
		#Title "Second Section of s1"
		#Description "Second Section Desc"

		@Question brahma {
			#Type TEXT
			#Title "What's your favorite Brahma?"
			@ShowIf { favoriteBeer equals "Brahma" }
		}
		
		@Question patagonia {
			#Type TEXT
			#Title "What's your favorite Patagonia?"
			@ShowIf { favoriteBeer equals "Patagonia" }
		}
		
		@ShowIf {
			(favoriteBeer equals "Brahma" OR favoriteBeer equals "Patagonia") AND howOld isGreaterThan 18
		}
	}
	
}

@Step s2 {
	#Title "Step 2"
	#Description "Step 2 desc"
	@Question s2q1 {
		#Type TEXT
		#Title "Step 2 question"
	}
}

@Step s3 {
	#Title "Step 3"
	#Description "Step 3 desc"
	@Question s3q1 {
		#Type DATE
		#Title "When were you born?"
		@Glitch {
			@Error {
				#Message "You are too old"
				@ShowIf { s3q1 isBefore DATE(15, 12, 2000) }
			}
		}
	}
}
